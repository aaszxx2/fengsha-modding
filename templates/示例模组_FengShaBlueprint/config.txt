# ---------- 导出 ----------
# 导出哪些建筑：固定为「玩家自己的全部建筑」（按阵营过滤）
# auto = 自动判定玩家阵营（取拥有建筑最多的那个阵营）
# 也可以直接写数字：1齐国 2楚国 3燕国 4韩国 5赵国 6魏国 7秦国 90中立 99裁判
ExportFaction        = auto

# 导出时是否把「施工中」的建筑也带上（默认只带已完工的）
IncludeConstructing  = false

# 单次导出建筑数量上限（防手滑导出一个几万栋的存档把分享码撑爆）
MaxBuildings         = 3000

# 蓝图名字；留空则自动用「基地_年月日_时分秒」
ExportName           =

# 导出时把分享码写进 Windows 剪贴板（方便直接粘贴给朋友）
UseClipboard         = true

# ---------- 导入 ----------
# 导入后是否直接完工。
#   true  = 立即完工，不消耗材料
#   false = 生成「工地」，交给游戏自己的民夫搬运管线施工 —— 会**自动按正常流程消耗材料**
#           （逆向结论：建造扣费不是一次扣费，而是「需求清单 → 民夫搬运 → 到位消耗」，
#             FastConstructBuildingActor 之所以不扣费，是因为它跳过了需求清单阶段）
InstantComplete      = true

# 旧版预留项，现已由 InstantComplete=false 覆盖，保留仅为兼容旧配置
ConsumeResources     = false

# 演练模式：true = 只解析分享码并把结果写进日志，不真的生成建筑
# 第一次用建议先开一次，确认解析没问题再关掉
DryRun               = false

# ---------- 界面 ----------
# 屏幕提示显示时长（毫秒）
HudMs                = 6000

# 是否输出详细日志（写进 ue4ss\UE4SS.log，用 [FengShaBP] 过滤）
Verbose              = true
